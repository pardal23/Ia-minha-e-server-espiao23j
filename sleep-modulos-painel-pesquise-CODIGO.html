/** * Super Função Módulos: Centraliza a lógica de busca avançada * @param {string} entradaUsuario - O termo digitado pelo programador */ function superFunctionModulos(entradaUsuario) { // Loop interativo de processamento (higienização de dados) let termosProcessados = []; const palavrasChave = entradaUsuario.split(' '); for (let i = 0; i < palavrasChave.length; i++) { let palavra = palavrasChave[i].trim().toLowerCase(); if (palavra.length > 0) { termosProcessados.push(palavra); } } // Contexto obrigatório focado em desenvolvimento web e engenharia de software const contextoWeb = "módulos programação desenvolvimento web arquitetura projetos backend frontend"; const queryFinal = termosProcessados.length > 0 ? termosProcessados.join(' ') + " " + contextoWeb : contextoWeb; // Executa a chamada externa aplicando a profundidade de 15 resultados pesquisarUniversalProfundidade15(queryFinal); } /** * Executa a pesquisa universal utilizando a API de busca do Google formatada * @param {string} query - Query completa combinada */ function pesquisarUniversalProfundidade15(query) { const googleBaseUrl = "https://google.com"; // num=15 define a profundidade da página para trazer 15 resultados estruturados const urlFinal = `${googleBaseUrl}?q=${encodeURIComponent(query)}&num=15`; // Abre a pesquisa profunda numa nova aba window.open(urlFinal, '_blank'); } // Escuta de eventos para ativação por teclado (Loop de captura do Usuário) document.getElementById('modulos-search').addEventListener('keydown', function(e) { if (e.key === 'z' || e.key === 'Z' || e.key === 'Enter') { e.preventDefault(); // Loop interativo: Valida se o campo não está vazio antes de disparar if (this.value.trim() === "") { alert("Por favor, digite um módulo ou tecnologia para pesquisar!"); this.focus(); return; } // Invoca a super função com o input do utilizador superFunctionModulos(this.value); } });
